USE CinemaBilletterie;
GO

CREATE OR ALTER PROCEDURE dbo.sp_GenererDonneesCinema
    @NbFilms   INT = 2000,
    @NbSalles  INT = 20,
    @NbSeances INT = 80000,
    @NbBillets INT = 50000000,
    @TailleLot INT = 500000
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    IF OBJECT_ID('dbo.Nombres') IS NULL
    BEGIN
        CREATE TABLE dbo.Nombres (N INT PRIMARY KEY);
    END

    IF (SELECT COUNT(*) FROM dbo.Nombres) < @TailleLot
    BEGIN
        TRUNCATE TABLE dbo.Nombres;
        ;WITH T1 AS (SELECT 1 AS N UNION ALL SELECT 1),
              T2 AS (SELECT 1 AS N FROM T1 A CROSS JOIN T1 B),
              T4 AS (SELECT 1 AS N FROM T2 A CROSS JOIN T2 B),
              T8 AS (SELECT 1 AS N FROM T4 A CROSS JOIN T4 B),
              T24 AS (SELECT ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS N FROM T8 A CROSS JOIN T8 B CROSS JOIN T8 C)
        INSERT INTO dbo.Nombres (N)
        SELECT TOP (@TailleLot) N FROM T24;
    END

    DELETE FROM t_billets;
    DELETE FROM t_seances;
    DELETE FROM t_salles;
    DELETE FROM t_Films;

    DBCC CHECKIDENT ('t_billets', RESEED, 0);
    DBCC CHECKIDENT ('t_seances', RESEED, 0);
    DBCC CHECKIDENT ('t_salles', RESEED, 0);
    DBCC CHECKIDENT ('t_Films', RESEED, 0);

    INSERT INTO t_Films (titre, annee, genre, dureeMin, realisateur)
    SELECT
        'Film ' + CAST(N AS VARCHAR(10)),
        2000 + (N % 25),
        (SELECT g FROM (VALUES ('Action'),('Comédie'),('Drame'),('SF'),('Animation'),('Horreur')) AS G(g)
           ORDER BY CHECKSUM(NEWID()) OFFSET 0 ROWS FETCH NEXT 1 ROWS ONLY),
        90 + (N % 60),
        'Réalisateur ' + CAST((N % 300) AS VARCHAR(10))
    FROM dbo.Nombres WHERE N <= @NbFilms;

    INSERT INTO t_salles (nom, capacite)
    SELECT 'Salle ' + CAST(N AS VARCHAR(5)), 80 + (N % 4) * 40
    FROM dbo.Nombres WHERE N <= @NbSalles;

    INSERT INTO t_seances (dateHeure, salles_id, films_id)
    SELECT
        DATEADD(MINUTE, N, '2026-01-01'),
        1 + (ABS(CHECKSUM(NEWID())) % @NbSalles),
        1 + (ABS(CHECKSUM(NEWID())) % @NbFilms)
    FROM dbo.Nombres WHERE N <= @NbSeances;

    DECLARE @i INT = 0;
    WHILE @i < @NbBillets
    BEGIN
        INSERT INTO t_billets (prix, dateAchat, seances_id)
        SELECT
            CAST(8 + (ABS(CHECKSUM(NEWID())) % 12) AS DECIMAL(5,2)),
            DATEADD(SECOND, ABS(CHECKSUM(NEWID())) % 5000000, '2026-01-01'),
            1 + (ABS(CHECKSUM(NEWID())) % @NbSeances)
        FROM dbo.Nombres
        WHERE N <= @TailleLot;

        SET @i += @TailleLot;
        PRINT CONCAT('Billets insérés : ', @i, ' / ', @NbBillets);
        CHECKPOINT;
    END

    SELECT
        (SELECT COUNT(*) FROM t_Films)   AS NbFilmsCrees,
        (SELECT COUNT(*) FROM t_salles)  AS NbSallesCrees,
        (SELECT COUNT(*) FROM t_seances) AS NbSeancesCrees,
        (SELECT COUNT(*) FROM t_billets) AS NbBilletsCrees;
END
GO

EXEC dbo.sp_GenererDonneesCinema;
GO
