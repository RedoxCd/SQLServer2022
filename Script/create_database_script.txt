CREATE DATABASE LootTable
COLLATE Latin1_General_CI_AS;
GO

USE LootTable;
GO

CREATE TABLE t_Jeux (
    jeux_id      INT IDENTITY,
    titre        VARCHAR(200) NOT NULL,
    plateforme   VARCHAR(50)  NOT NULL,
    genre        VARCHAR(50)  NOT NULL,
    prix         DECIMAL(5,2) NOT NULL,
    editeur      VARCHAR(100) NOT NULL,
    anneeSortie  INT NOT NULL,
    PRIMARY KEY (jeux_id)
);
CREATE TABLE t_Joueurs (
    joueurs_id     BIGINT IDENTITY,
    pseudo         VARCHAR(50) NOT NULL,
    dateCreation   DATETIME2 NOT NULL,
    PRIMARY KEY (joueurs_id)
);
CREATE TABLE t_Achats (
    achats_id    BIGINT IDENTITY,
    joueurs_id   BIGINT NOT NULL,
    jeux_id      INT NOT NULL,
    prixPaye     DECIMAL(5,2) NOT NULL,
    dateAchat    DATETIME2 NOT NULL,
    offertPar    VARCHAR(50) NULL,  -- pseudo/nom du visiteur qui offre
    PRIMARY KEY (achats_id),
    FOREIGN KEY (joueurs_id) REFERENCES t_Joueurs(joueurs_id),
    FOREIGN KEY (jeux_id) REFERENCES t_Jeux(jeux_id)
);
